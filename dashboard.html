// Normalización de pasajes activos
pasajesData = rawPasajes.map(d => {
  const precio = +d.precio_bob || 0;
  const precioMin = +d.precio_minimo_bob || precio;
  const origen = (d.origen_nombre || "").toString().trim().toUpperCase();
  const destino = (d.destino_nombre || "").toString().trim().toUpperCase();
  const tipo = (d.tipo_transporte || "").toString().trim().toUpperCase();
  const empresa = (d.empresa_aerolinea || "").toString().trim();
  const ruta = `${origen} ➔ ${destino}`;

  return {
    ...d,
    tipo_transporte: tipo,
    origen_nombre: origen,
    destino_nombre: destino,
    empresa_aerolinea: empresa,
    ruta: ruta,
    precio_bob: precio,
    precio_minimo_bob: precioMin,
    categoria_cabina: (d.categoria_cabina || "").toString().trim() || "Estándar"
  };
}).filter(d => d.origen_nombre && d.destino_nombre)

// Normalización de resumen agregado de rutas
rutasData = rawResumenRutas.map(d => {
  return {
    ...d,
    precio_min: +d.precio_min || 0,
    precio_avg: +d.precio_avg || 0,
    precio_max: +d.precio_max || 0,
    total_ofertas: +d.total_ofertas || 0
  };
})

// Normalización de FIDS
fidsData = rawFids.map(d => {
  const aero = (d.AEROPUERTO || "").toString().trim();
  const tipoOp = (d.TIPO_OPERACION || "").toString().trim().toUpperCase() === "S" ? "Salida" : "Llegada";
  const aerolinea = (d.NOMBRE_AEROLINEA || "").toString().trim();
  const obs = (d.OBSERVACION || "").toString().trim() || "Normal";
  
  return {
    ...d,
    AEROPUERTO: aero,
    tipo_operacion_lbl: tipoOp,
    NOMBRE_AEROLINEA: aerolinea,
    OBSERVACION: obs
  };
}).filter(d => d.AEROPUERTO)

// Listas únicas para filtros
tiposTransporte = ["Todos", "VUELO", "BUS"]
origenesList = ["Todos"].concat(Array.from(new Set(pasajesData.map(d => d.origen_nombre))).sort())
destinosList = ["Todos"].concat(Array.from(new Set(pasajesData.map(d => d.destino_nombre))).sort())
empresasList = ["Todas"].concat(Array.from(new Set(pasajesData.map(d => d.empresa_aerolinea))).filter(Boolean).sort())
aeropuertosFidsList = ["Todos"].concat(Array.from(new Set(fidsData.map(d => d.AEROPUERTO))).sort())

maxPrecioPosible = d3.max(pasajesData, d => d.precio_bob) || 2000

filteredPasajes = pasajesData.filter(d => {
  const matchTipo = selectedTipo === "Todos" || d.tipo_transporte === selectedTipo;
  const matchOrigen = selectedOrigen === "Todos" || d.origen_nombre === selectedOrigen;
  const matchDestino = selectedDestino === "Todos" || d.destino_nombre === selectedDestino;
  const matchEmpresa = selectedEmpresa === "Todas" || d.empresa_aerolinea === selectedEmpresa;
  const matchPrecio = d.precio_bob <= maxPrecioFiltro;
  
  let matchSearch = true;
  if (searchFiltro && searchFiltro.trim() !== "") {
    const q = searchFiltro.toLowerCase();
    matchSearch = (
      (d.empresa_aerolinea && d.empresa_aerolinea.toLowerCase().includes(q)) ||
      (d.ruta && d.ruta.toLowerCase().includes(q)) ||
      (d.numero_vuelo_bus && d.numero_vuelo_bus.toString().toLowerCase().includes(q))
    );
  }

  return matchTipo && matchOrigen && matchDestino && matchEmpresa && matchPrecio && matchSearch;
})
